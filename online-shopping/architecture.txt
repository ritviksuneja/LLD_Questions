Online Shopping Service (LLD)
Problem Statement
Design and implement an Online Shopping Service that allows users to browse products, add items to a cart, place orders, make payments, and track order status.

Requirements
User Management: Users can register, log in, and manage their profiles.
Product Catalog: The system manages a catalog of products with details and prices.
Cart Management: Users can add, update, or remove products in their shopping cart.
Order Placement: Users can place orders for products in their cart.
Order Tracking: The system tracks the status of each order (e.g., PLACED, SHIPPED, DELIVERED, CANCELLED).
Payment Processing: Users can pay for their orders using different payment methods.
Extensibility: Easy to add new features such as discounts, reviews, or wishlists.

enum PaymentMethodType {
    CARD, UPI
}

class Order {
    int id;
    double amount;
    PaymentMethodType paymentMethod;  // how this order will be paid

    Order(int id, double amount, PaymentMethodType paymentMethod) {
        this.id = id;
        this.amount = amount;
        this.paymentMethod = paymentMethod;
    }

    public PaymentMethodType getPaymentMethod() {
        return paymentMethod;
    }
}


interface PaymentStrategy {
    void pay(Order order);
}

class CardPaymentStrategy implements PaymentStrategy {
    @Override
    public void pay(Order order) {
        System.out.println("Processing CARD payment for order " +
                order.id + " of amount " + order.amount);
    }
}

class UpiPaymentStrategy implements PaymentStrategy {
    @Override
    public void pay(Order order) {
        System.out.println("Processing UPI payment for order " +
                order.id + " of amount " + order.amount);
    }
}

class PaymentProcessor {
    public void process(Order order) {
        PaymentStrategy strategy;

        // choose strategy based on order.paymentMethod
        if (order.getPaymentMethod() == PaymentMethodType.CARD) {
            strategy = new CardPaymentStrategy();
        } else {
            strategy = new UpiPaymentStrategy();
        }

        strategy.pay(order);
    }
}

public class Demo {
    public static void main(String[] args) {
        Order order = new Order(1, 500.0, PaymentMethodType.CARD);

        PaymentProcessor processor = new PaymentProcessor();
        processor.process(order);  // internally picks CARD strategy
    }
}
